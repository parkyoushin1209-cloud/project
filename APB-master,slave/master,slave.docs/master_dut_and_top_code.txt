
package APB_ifc
typedef enum logic[1:0] {IDLE, SETUP, ACCESS} state_t;

endpackage

interface APB_ifc(input logic PCLK, PRESETn);
// master <=> requester
//PADDR
// slave  <=> completer

parameter DATA_WIDTH = 8;
parameter ADDR_WIDTH = 16;
parameter SLV_num = 4;

logic PNSE, PENABLE, PWRITE, PREADY;
logic[2:0] PPROT;
logic [DATA_WIDTH-1:0] PWDATA, PRDATA;
logic[ADDR_WIDTH-1:0] PADDR;
logic[SLV_num-1:0] PSELx;


modport master(output PADDR, PPROT, PNSE, PSELx, PENABLE, PWRITE, PWDATA
               input PREADY, PRDATA, PCLK, PRESETn);

modport slave(output PREADY, PRDATA, PCLK, PRESETn,
              input PADDR, PPROT, PNSE, PSELx, PENABLE, PWRITE, PWDATA);

endinterface


module top;
`include "uvm_macros.svh"
import uvm_pkg::*;
import APB_ifc::*;

logic PCLK, PRESETn;

initial begin 
    PRESETn = 1;
    #2 PRESETn = 0;
    #2 PRESETn = 1;
end

clk_gen clk(PCLK);
APB_ifc ifc(PCLK, PRESETn);
master mst(ifc.master);

initial begin
uvm_config_db #(virtual ifc) :: set(null, "*", "ifc", ifc);
run_test();
end
endmodule


module clk_gen(output logic PCLK);
logic local_clk;
assign PCLK = local_clk;

always #5 local_clk = ~local_clk;
endmodule

module master(input APB_ifc.master ifc);
state_t state, next_state;

always @(posedge ifc.PCLK or negedge ifc.PRESETn) begin
if(!ifc.PRESETn) begin 
    state <= IDLE; 
    ifc.PNSE <= 0;
    ifc.PENABLE <= 0;
    ifc.PWRITE <= 0;
    ifc.PREADY <= 0;
    ifc.PPROT <= 3'b000;
    ifc.PWDATA <= 0;
    ifc.PRDATA <= 0;
    ifc.PADDR <= 0;
    ifc.PSELx <= 0;
    end
else state <= next_state;
end

always @(*) begin // always(*)에서는 블로킹 사용하기
    case(state)
    IDLE : begin (ifc.is_there_next_transfer) ? (next_state = SETUP) : (next_state = IDLE);
    if(ifc.is_there_next_transfer) begin 
        if (ifc.readwrite)       // read
            ifc.PWRITE = 0;
        else                     // write
            ifc.PWRITE = 1;

        ifc.PADDR = next_transfer_addr;
        ifc.PSELx = 1;
        ifc.PENABLE = 0; 
    end
    end

    SETUP : begin
    assert(ifc.PSELx && !ifc.PENABLE)  // 어써션은 시뮬레이션에서만
    else $fatal("APB SETUP phase error: PENABLE should be 0 when PSELx is 1");
    next_state = ACCESS;
    end

    ACCESS : begin
    assert(ifc.PSELx && ifc.PENABLE)
    else $fatal("APB ACCESS phase error: PENABLE and PSELx should be 1");
    (ifc.PREADY) ? ((ifc.is_there_next_transfer) ? (next_state = SETUP) : (next_state = IDLE)) : (next_state = ACCESS);
    end
    endcase
end

endmodule
